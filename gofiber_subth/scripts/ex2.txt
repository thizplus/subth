5. ปรับ BATCH_SIZE และ Safety Settings
เนื่องจากงานนี้เป็นแนว Adult (แม้จะใช้ภาษาไทยกวี) Gemini อาจจะ Block เนื้อหาบางส่วน (Safety Filter)

Python
# เพิ่มตอน setup_gemini
def setup_gemini():
    genai.configure(api_key=GEMINI_API_KEY)
    # ปรับ Safety Settings เพื่อไม่ให้ Block คำว่า 'กำหนัด' หรือ 'สวาท'
    safety_settings = [
        {"category": "HARM_CATEGORY_SEXUALLY_EXPLICIT", "threshold": "BLOCK_NONE"},
        {"category": "HARM_CATEGORY_HARASSMENT", "threshold": "BLOCK_NONE"},
    ]
    return genai.GenerativeModel('gemini-2.0-flash', safety_settings=safety_settings)
